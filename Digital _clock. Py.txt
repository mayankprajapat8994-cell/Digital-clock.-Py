import tkinter as tk
from datetime import datetime

# Create window
root = tk.Tk()
root.title("Advanced Digital Clock")
root.geometry("900x550")
root.configure(bg="#101015")

# Fullscreen
fullscreen = False

# Colors
bg = "#101015"
white = "#F5F5F5"
blue = "#4AAFFF"
green = "#50DC96"
gray = "#999999"

# Title
title = tk.Label(
    root,
    text="ADVANCED DIGITAL CLOCK",
    font=("Arial", 28, "bold"),
    bg=bg,
    fg=blue
)
title.pack(pady=30)

# Time
time_label = tk.Label(
    root,
    font=("Consolas", 90, "bold"),
    bg=bg,
    fg=white
)
time_label.pack(pady=10)

# AM/PM
ampm_label = tk.Label(
    root,
    font=("Arial", 32, "bold"),
    bg=bg,
    fg=green
)
ampm_label.pack()

# Date
date_label = tk.Label(
    root,
    font=("Arial", 28),
    bg=bg,
    fg=gray
)
date_label.pack(pady=15)

# Day
day_label = tk.Label(
    root,
    font=("Arial", 40, "bold"),
    bg=bg,
    fg=blue
)
day_label.pack()

# Status
status_label = tk.Label(
    root,
    text="● CLOCK RUNNING",
    font=("Arial", 20, "bold"),
    bg=bg,
    fg=green
)
status_label.pack(pady=25)

# Update clock
def update_clock():
    now = datetime.now()

    time_label.config(
        text=now.strftime("%I:%M:%S")
    )

    ampm_label.config(
        text=now.strftime("%p")
    )

    date_label.config(
        text=now.strftime("%d %B %Y")
    )

    day_label.config(
        text=now.strftime("%A")
    )

    root.after(1000, update_clock)

# Fullscreen
def fullscreen_mode(event=None):
    global fullscreen
    fullscreen = not fullscreen
    root.attributes("-fullscreen", fullscreen)

# Exit
def exit_clock(event=None):
    root.destroy()

# Keyboard controls
root.bind("<F11>", fullscreen_mode)
root.bind("<Escape>", exit_clock)

# Start clock
update_clock()

root.mainloop()
